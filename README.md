  <caption>un mec</caption>
  <thead><tr>
   <th>Nom</th>
   <th>Age</th>
   <th>metier</th>
  </thead></tr>
 <tbody><tr>
   <td>Clement</td>
   <td>25</td>
   <td>codeur</td>
 </tbody></tr<>

</table>
